# 网页播放器（Android）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个安卓 App：粘贴网页链接后嗅探页面视频（HLS/MP4/WebM），优先用原生播放器播放，提取失败时回退到内置网页播放，并提供历史、收藏与续播。

**Architecture:** 单模块 Kotlin + Jetpack Compose。`MainActivity` 承载首页与 WebView 网页页；`PlayerActivity` 独立承载 Media3 播放与画中画。WebView 通过 `shouldInterceptRequest`、注入 JS 钩子、页面源码正则三层嗅探媒体；播放时透传 WebView 的 UA/Referer/Origin/Cookie。数据用 Room + DataStore 本地存储。

**Tech Stack:** Kotlin 2.0.21、AGP 8.7.3、Gradle 8.10.2、Compose BOM 2024.12.01、Media3 1.5.0、Room 2.6.1、DataStore 1.1.1、OkHttp 4.12.0、JUnit4。

**Spec:** `docs/superpowers/specs/2026-11-07-网页视频播放器-design.md`（执行时必须同时阅读此规格）

```kotlin
package com.jqc.webplayer.core

import java.net.URI
import java.net.URISyntaxException

object UrlNormalizer {
    private val schemeRegex = Regex("^[a-zA-Z][a-zA-Z0-9+.-]*://")

    fun normalize(raw: String): String? {
        val trimmed = raw.trim()
        if (trimmed.isEmpty() || trimmed.length > 4096) return null
        val withScheme = if (schemeRegex.containsMatchIn(trimmed)) trimmed else "https://$trimmed"
        val lower = withScheme.lowercase()
        if (!lower.startsWith("http://") && !lower.startsWith("https://")) return null
        return try {
            val uri = URI(withScheme)
            if (uri.host.isNullOrBlank()) null else withScheme
        } catch (e: URISyntaxException) {
            null
        }
    }
}
```

`MediaUrlClassifier.kt`：

```kotlin
package com.jqc.webplayer.core

import java.net.URI

object MediaUrlClassifier {
    private val blockedSuffixes = listOf(
        ".ts", ".m4s", ".aac", ".mp3", ".vtt", ".key",
        ".jpg", ".jpeg", ".png", ".gif", ".webp", ".css", ".js", ".json",
    )
    private val blockedFragments = listOf(
        "/ad/", "/ads/", "advert", "doubleclick", "googlesyndication", "/beacon", "analytics",
    )
    private val playHints = listOf("play", "stream", "video", "media", "hls")

    fun detectType(rawUrl: String): StreamType? {
        val url = rawUrl.trim()
        if (url.length > 4096) return null
        val lower = url.lowercase()
        if (!lower.startsWith("http://") && !lower.startsWith("https://")) return null
        val path = url.substringBefore('?').substringBefore('#').lowercase()
        return when {
            path.endsWith(".m3u8") -> StreamType.HLS
            path.endsWith(".mp4") -> StreamType.MP4
            path.endsWith(".webm") -> StreamType.WEBM
            else -> null
        }
    }

    fun classify(
        rawUrl: String,
        pageUrl: String?,
        source: SniffSource,
        now: Long = System.currentTimeMillis(),
    ): MediaCandidate? {
        val url = rawUrl.trim()
        val type = detectType(url) ?: return null
        val lower = url.lowercase()
        val path = url.substringBefore('?').substringBefore('#').lowercase()
        if (blockedSuffixes.any { path.endsWith(it) }) return null
        if (blockedFragments.any { lower.contains(it) }) return null
        val host = hostOf(url) ?: return null
        return MediaCandidate(
            url = url,
            type = type,
            host = host,
            qualityLabel = null,
            discoveredAt = now,
            source = source,
        )
    }

    fun rank(candidates: List<MediaCandidate>, pageHost: String?): List<MediaCandidate> {
        val deduped = LinkedHashMap<String, MediaCandidate>()
        candidates.forEach { deduped[it.url] = it }
        return deduped.values.sortedWith(
            compareByDescending<MediaCandidate> { score(it, pageHost) }
                .thenByDescending { it.discoveredAt }
        )
    }

    fun hostOf(url: String): String? = try {
        URI(url).host
    } catch (e: Exception) {
        null
    }

    private fun score(candidate: MediaCandidate, pageHost: String?): Int {
        var score = when (candidate.type) {
            StreamType.HLS -> 40
            StreamType.MP4 -> 30
            StreamType.WEBM -> 25
        }
        if (pageHost != null && candidate.host.equals(pageHost, ignoreCase = true)) score += 3
        val path = candidate.url.substringBefore('?').lowercase()
        if (playHints.any { path.contains(it) }) score += 2
        return score
    }
}
```

```kotlin
package com.jqc.webplayer.core

class MediaCollector(private val clock: () -> Long = System::currentTimeMillis) {
    private val lock = Any()
    private var pageUrl: String? = null
    private val candidates = LinkedHashMap<String, MediaCandidate>()

    fun startPage(url: String) = synchronized(lock) {
        pageUrl = url
        candidates.clear()
    }

    fun updatePage(url: String) = synchronized(lock) {
        pageUrl = url
    }

    fun offer(rawUrl: String, source: SniffSource): List<MediaCandidate> = synchronized(lock) {
        val candidate = MediaUrlClassifier.classify(rawUrl, pageUrl, source, clock())
        if (candidate != null && !candidates.containsKey(candidate.url)) {
            candidates[candidate.url] = candidate
            if (candidates.size > 50) {
                val oldest = candidates.values.minByOrNull { it.discoveredAt }
                if (oldest != null) candidates.remove(oldest.url)
            }
        }
        snapshotLocked()
    }

    fun setQuality(url: String, label: String): List<MediaCandidate> = synchronized(lock) {
        candidates[url]?.let { candidates[url] = it.copy(qualityLabel = label) }
        snapshotLocked()
    }

    fun snapshot(): List<MediaCandidate> = synchronized(lock) { snapshotLocked() }

    fun clear() = synchronized(lock) { candidates.clear() }

    private fun snapshotLocked(): List<MediaCandidate> {
        val pageHost = pageUrl?.let { MediaUrlClassifier.hostOf(it) }
        return MediaUrlClassifier.rank(candidates.values.toList(), pageHost)
    }
}
```

```kotlin
package com.jqc.webplayer.core

import java.net.URI

data class HlsVariant(
    val url: String,
    val width: Int?,
    val height: Int?,
    val bandwidth: Long,
)

object HlsManifestParser {
    private val resolutionRegex = Regex("RESOLUTION=(\\d+)x(\\d+)")
    private val bandwidthRegex = Regex("BANDWIDTH=(\\d+)")

    fun isMaster(text: String): Boolean =
        text.lineSequence().any { it.trim().startsWith("#EXT-X-STREAM-INF") }

    fun parseMaster(text: String, baseUrl: String): List<HlsVariant> {
        val lines = text.lineSequence().map { it.trim() }.filter { it.isNotEmpty() }.toList()
        val result = mutableListOf<HlsVariant>()
        var width: Int? = null
        var height: Int? = null
        var bandwidth = 0L
        var awaitingUri = false

        for (line in lines) {
            if (line.startsWith("#EXT-X-STREAM-INF")) {
                val attrs = line.substringAfter(':', "")
                val resolution = resolutionRegex.find(attrs)
                width = resolution?.groupValues?.get(1)?.toIntOrNull()
                height = resolution?.groupValues?.get(2)?.toIntOrNull()
                bandwidth = bandwidthRegex.find(attrs)?.groupValues?.get(1)?.toLongOrNull() ?: 0L
                awaitingUri = true
            } else if (!line.startsWith("#") && awaitingUri) {
                val resolved = resolve(baseUrl, line)
                if (resolved != null) {
                    result += HlsVariant(resolved, width, height, bandwidth)
                }
                awaitingUri = false
            }
        }
        return result
    }

    fun qualityLabel(variants: List<HlsVariant>): String? =
        variants.maxByOrNull { it.height ?: 0 }?.height?.let { "${it}P" }

    private fun resolve(baseUrl: String, reference: String): String? = try {
        URI(baseUrl).resolve(reference).toString()
    } catch (e: Exception) {
        null
    }
}
```

```kotlin
package com.jqc.webplayer.core

import java.net.URI

object PlaybackHeaders {
    fun build(
        pageUrl: String,
        userAgent: String,
        videoCookies: String?,
        pageCookies: String?,
    ): Map<String, String> {
        val headers = linkedMapOf<String, String>()
        headers["User-Agent"] = userAgent
        headers["Accept"] = "*/*"
        if (pageUrl.isNotBlank()) headers["Referer"] = pageUrl
        originOf(pageUrl)?.let { headers["Origin"] = it }

        val cookies = (listOfNotNull(videoCookies, pageCookies))
            .flatMap { it.split(";") }
            .map { it.trim() }
            .filter { it.isNotEmpty() && it.contains("=") }
            .distinct()
        if (cookies.isNotEmpty()) headers["Cookie"] = cookies.joinToString("; ")

        return headers
    }

    fun originOf(pageUrl: String): String? = try {
        val uri = URI(pageUrl)
        val host = uri.host
        if (host.isNullOrBlank()) null else buildString {
            append(uri.scheme)
            append("://")
            append(host)
            if (uri.port > 0) append(":").append(uri.port)
        }
    } catch (e: Exception) {
        null
    }
}
```

```kotlin
class HlsProbe(private val client: OkHttpClient = OkHttpClient()) {

        withContext(Dispatchers.IO) {
            runCatching {
                val request = Request.Builder()
                    .url(candidate.url)
                    .apply { headers.forEach { (key, value) -> header(key, value) } }
                    .build()
                client.newCall(request).execute().use { response ->
                    if (!response.isSuccessful) return@use null
                    val text = response.peekBody(262_144).string()
                    val variants = com.jqc.webplayer.core.HlsManifestParser.parseMaster(text, candidate.url)
                    com.jqc.webplayer.core.HlsManifestParser.qualityLabel(variants)
                }
            }.getOrNull()
        }
}
```

`TimeFormat.kt`：

```kotlin
package com.jqc.webplayer.ui.common

import java.time.Instant
import java.time.ZoneId
import java.time.format.DateTimeFormatter
import java.util.Locale

object TimeFormat {
    fun duration(ms: Long): String {
        val totalSeconds = (ms.coerceAtLeast(0) + 500) / 1000
        val hours = totalSeconds / 3600
        val minutes = (totalSeconds % 3600) / 60
        val seconds = totalSeconds % 60
        return if (hours > 0) {
            String.format(Locale.US, "%d:%02d:%02d", hours, minutes, seconds)
        } else {
            String.format(Locale.US, "%02d:%02d", minutes, seconds)
        }
    }

    fun relative(timestampMs: Long, nowMs: Long, zone: ZoneId = ZoneId.systemDefault()): String {
        val diff = nowMs - timestampMs
        return when {
            diff < 60_000 -> "刚刚"
            diff < 3_600_000 -> "${diff / 60_000} 分钟前"
            diff < 86_400_000 -> "${diff / 3_600_000} 小时前"
            else -> {
                val date = Instant.ofEpochMilli(timestampMs).atZone(zone).toLocalDate()
                val now = Instant.ofEpochMilli(nowMs).atZone(zone).toLocalDate()
                val pattern = if (date.year == now.year) "MM-dd" else "yyyy-MM-dd"
                date.format(DateTimeFormatter.ofPattern(pattern))
            }
        }
    }
}
```

```kotlin
package com.jqc.webplayer.ui.common

import android.content.ClipboardManager
import android.content.Context
import androidx.compose.runtime.Composable
import androidx.compose.runtime.DisposableEffect
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.LifecycleEventObserver
import androidx.lifecycle.compose.LocalLifecycleOwner
import com.jqc.webplayer.core.UrlNormalizer

private fun readClipboardUrl(context: Context): String? = runCatching {
    val manager = context.getSystemService(ClipboardManager::class.java) ?: return@runCatching null
    val clip = manager.primaryClip ?: return@runCatching null
    if (clip.itemCount == 0) return@runCatching null
    UrlNormalizer.normalize(clip.getItemAt(0).coerceToText(context).toString())
}.getOrNull()

@Composable
fun rememberClipboardUrl(): String? {
    val context = androidx.compose.ui.platform.LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current
    var url by remember { mutableStateOf(readClipboardUrl(context)) }

    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            if (event == Lifecycle.Event.ON_RESUME) {
                url = readClipboardUrl(context)
            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
    }
    return url
}
```

```kotlin
import android.content.Context

data class PlaybackRequest(
    val url: String,
    val type: StreamType,
    val pageUrl: String,
    val title: String,
    val headers: Map<String, String>,
) {
    fun toIntent(context: Context): Intent = Intent(context, PlayerActivity::class.java).apply {
        putExtra(PlayerContract.EXTRA_URL, url)
        putExtra(PlayerContract.EXTRA_TYPE, type.name)
        putExtra(PlayerContract.EXTRA_PAGE_URL, pageUrl)
        putExtra(PlayerContract.EXTRA_TITLE, title)
        putExtra(
            PlayerContract.EXTRA_HEADERS,
            Bundle().apply { headers.forEach { (key, value) -> putString(key, value) } }
        )
    }
}

object PlayerContract {
    const val EXTRA_URL = "extra_url"
    const val EXTRA_TYPE = "extra_type"
    const val EXTRA_PAGE_URL = "extra_page_url"
    const val EXTRA_TITLE = "extra_title"
    const val EXTRA_HEADERS = "extra_headers"
    const val RESULT_OPEN_WEB = 1001

    fun headersFrom(intent: Intent): Map<String, String> {
        val bundle = intent.getBundleExtra(EXTRA_HEADERS) ?: return emptyMap()
        return bundle.keySet().associateWith { bundle.getString(it).orEmpty() }
    }
}
```

```javascript
(function () {
    if (window.__jqcSnifferInstalled) return;
    window.__jqcSnifferInstalled = true;

    function report(u, hint) {
        try {
            if (!u || typeof u !== 'string') return;
            if (!/^https?:\/\//i.test(u)) return;
            if (u.length > 4096) return;
            if (window.JqcBridge && window.JqcBridge.reportMedia) {
                window.JqcBridge.reportMedia(u, hint || '');
            }
        } catch (e) {}
    }

    function absolute(u) {
        try { return new URL(u, location.href).href; } catch (e) { return null; }
    }

    var originalFetch = window.fetch;
    if (originalFetch) {
        window.fetch = function (input) {
            try {
                var raw = typeof input === 'string' ? input : (input && input.url);
                var abs = raw ? absolute(raw) : null;
                if (abs) report(abs, 'fetch');
            } catch (e) {}
            var promise = originalFetch.apply(this, arguments);
            if (promise && promise.then) {
                promise.then(function (response) {
                    try { if (response && response.url) report(response.url, 'fetch'); } catch (e) {}
                });
            }
            return promise;
        };
    }

    var originalOpen = XMLHttpRequest.prototype.open;
    XMLHttpRequest.prototype.open = function (method, url) {
        try {
            var abs = absolute(url);
            if (abs) report(abs, 'xhr');
        } catch (e) {}
        return originalOpen.apply(this, arguments);
    };

    var originalSend = XMLHttpRequest.prototype.send;
    XMLHttpRequest.prototype.send = function () {
        var xhr = this;
        xhr.addEventListener('load', function () {
            try { if (xhr.responseURL) report(xhr.responseURL, 'xhr'); } catch (e) {}
        });
        return originalSend.apply(this, arguments);
    };

    function scan() {
        try {
            var nodes = document.querySelectorAll('video, source');
            for (var i = 0; i < nodes.length; i++) {
                var el = nodes[i];
                if (el.currentSrc) report(el.currentSrc, 'video');
                if (el.src) report(el.src, 'video');
            }
            var html = document.documentElement ? document.documentElement.innerHTML : '';
            if (html) {
                var re = /https?:(?:\\?\/){2}[^"'\s<>]+?\.m3u8[^"'\s<>]*/g;
                var match;
                var guard = 0;
                while ((match = re.exec(html)) && guard++ < 20) {
                    report(match[0].replace(/\\\//g, '/'), 'html');
                }
            }
        } catch (e) {}
    }

    var scheduled = false;
    function scheduleScan() {
        if (scheduled) return;
        scheduled = true;
        setTimeout(function () {
            scheduled = false;
            scan();
        }, 300);
    }

    document.addEventListener('DOMContentLoaded', scan);
    try {
        new MutationObserver(scheduleScan).observe(document.documentElement || document, {
            childList: true,
            subtree: true,
        });
    } catch (e) {}
    scan();
})();
```

```kotlin
package com.jqc.webplayer.browser

import android.annotation.SuppressLint
import android.app.Activity
import android.content.pm.ActivityInfo
import android.graphics.Color
import android.net.Uri
import android.view.View
import android.view.ViewGroup
import android.view.WindowManager
import android.webkit.CookieManager
import android.webkit.GeolocationPermissions
import android.webkit.JavascriptInterface
import android.webkit.PermissionRequest
import android.webkit.RenderProcessGoneDetail
import android.webkit.SslErrorHandler
import android.webkit.WebChromeClient
import android.webkit.WebResourceError
import android.webkit.WebResourceRequest
import android.webkit.WebResourceResponse
import android.webkit.WebSettings
import android.webkit.WebView
import android.webkit.WebViewClient
import android.widget.FrameLayout
import androidx.core.view.WindowCompat
import androidx.core.view.WindowInsetsCompat
import androidx.core.view.WindowInsetsControllerCompat
import com.jqc.webplayer.core.MediaCandidate
import com.jqc.webplayer.core.MediaCollector
import com.jqc.webplayer.core.MediaUrlClassifier
import com.jqc.webplayer.core.SniffSource
import com.jqc.webplayer.data.AppSettings

@SuppressLint("SetJavaScriptEnabled")
class BrowserController(
    private val activity: Activity,
    private val settings: AppSettings,
    private val collector: MediaCollector,
    private val listener: Listener,
) {
    interface Listener {
        fun onCandidates(candidates: List<MediaCandidate>)
        fun onTitleChanged(title: String)
        fun onUrlChanged(url: String)
        fun onPageError(message: String?)
        fun onSslError(url: String)
        fun onRenderProcessGone()
        fun onPageFinished()
        fun onMessage(message: String)
    }

    val webView: WebView = WebView(activity)

    private var desktopUa: Boolean = settings.desktopUa
    private var blockPopups: Boolean = settings.blockPopups
    private var fullscreenView: View? = null
    private var fullscreenCallback: WebChromeClient.CustomViewCallback? = null
    private var fullscreenContainer: FrameLayout? = null
    private var previousOrientation: Int = activity.requestedOrientation

    private val snifferScript: String by lazy {
        activity.assets.open("sniffer.js").bufferedReader().use { it.readText() }
    }

    init {
        val cookieManager = CookieManager.getInstance()
        cookieManager.setAcceptCookie(true)
        cookieManager.setAcceptThirdPartyCookies(webView, true)

        webView.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true
            databaseEnabled = true
            javaScriptCanOpenWindowsAutomatically = !settings.blockPopups
            setSupportMultipleWindows(true)
            mediaPlaybackRequiresUserGesture = true
            mixedContentMode = WebSettings.MIXED_CONTENT_NEVER_ALLOW
            useWideViewPort = true
            loadWithOverviewMode = true
            builtInZoomControls = false
            displayZoomControls = false
        }
        applyUserAgent()

        webView.webViewClient = object : WebViewClient() {
            override fun shouldOverrideUrlLoading(view: WebView, request: WebResourceRequest): Boolean {
                val scheme = request.url.scheme?.lowercase()
                if (scheme == "http" || scheme == "https") return false
                listener.onMessage("已拦截外部链接：$scheme")
                return true
            }

            override fun shouldInterceptRequest(view: WebView, request: WebResourceRequest): WebResourceResponse? {
                val url = request.url.toString()
                if (MediaUrlClassifier.detectType(url) != null) {
                    val list = collector.offer(url, SniffSource.REQUEST)
                    activity.runOnUiThread { listener.onCandidates(list) }
                }
                return null
            }

            override fun onPageStarted(view: WebView, url: String, favicon: android.graphics.Bitmap?) {
                super.onPageStarted(view, url, favicon)
                collector.startPage(url)
                injectSniffer()
                listener.onUrlChanged(url)
                listener.onCandidates(collector.snapshot())
                listener.onPageError(null)
            }

            override fun doUpdateVisitedHistory(view: WebView, url: String, isReload: Boolean) {
                super.doUpdateVisitedHistory(view, url, isReload)
                collector.updatePage(url)
                listener.onUrlChanged(url)
                listener.onCandidates(collector.snapshot())
            }

            override fun onPageFinished(view: WebView, url: String) {
                super.onPageFinished(view, url)
                injectSniffer()
                listener.onTitleChanged(view.title.orEmpty())
                listener.onPageFinished()
            }

            override fun onReceivedError(view: WebView, request: WebResourceRequest, error: WebResourceError) {
                super.onReceivedError(view, request, error)
                if (request.isForMainFrame) {
                    listener.onPageError("加载失败：${error.description}。请检查网络后重试。")
                }
            }

            override fun onReceivedSslError(view: WebView, handler: SslErrorHandler, error: android.net.http.SslError) {
                handler.cancel()
                listener.onSslError(error.url.orEmpty())
            }

            override fun onRenderProcessGone(view: WebView, detail: RenderProcessGoneDetail): Boolean {
                listener.onRenderProcessGone()
                return true
            }
        }

        webView.webChromeClient = object : WebChromeClient() {
            override fun onReceivedTitle(view: WebView, title: String?) {
                listener.onTitleChanged(title.orEmpty())
            }

            override fun onShowCustomView(view: View?, callback: CustomViewCallback?) {
                if (view == null || callback == null) return
                enterFullscreen(view, callback)
            }

            override fun onHideCustomView() {
                exitFullscreen()
            }

            override fun onCreateWindow(
                view: WebView,
                isDialog: Boolean,
                isUserGesture: Boolean,
                resultMsg: android.os.Message,
            ): Boolean {
                if (blockPopups) return false
                val transport = resultMsg.obj as? WebView.WebViewTransport ?: return false
                // 不弹新窗口：把弹窗的目标地址接回当前 WebView。
                val popup = WebView(activity).apply {
                    settings.javaScriptEnabled = true
                    webViewClient = object : WebViewClient() {
                        override fun shouldOverrideUrlLoading(
                            popupView: WebView,
                            request: WebResourceRequest,
                        ): Boolean {
                            val target = request.url.toString()
                            if (target.startsWith("http://") || target.startsWith("https://")) {
                                webView.loadUrl(target)
                            }
                            popupView.post { popupView.destroy() }
                            return true
                        }
                    }
                }
                transport.webView = popup
                resultMsg.sendToTarget()
                return true
            }

            override fun onPermissionRequest(request: PermissionRequest) {
                request.deny()
            }

            override fun onGeolocationPermissionsShowPrompt(
                origin: String,
                callback: GeolocationPermissions.Callback,
            ) {
                callback.invoke(origin, false, false)
            }
        }

        webView.addJavascriptInterface(Bridge(), "JqcBridge")
        webView.setDownloadListener { _, _, _, _, _ ->
            listener.onMessage("第一版不支持下载")
        }
    }

    fun load(url: String) {
        webView.loadUrl(url)
    }

    fun reload() {
        webView.reload()
    }

    fun goBack() {
        if (webView.canGoBack()) webView.goBack()
    }

    fun canGoBack(): Boolean = webView.canGoBack()

    fun currentUrl(): String = webView.url.orEmpty()

    fun userAgent(): String = webView.settings.userAgentString ?: mobileUserAgent()

    fun setDesktopUa(enabled: Boolean) {
        if (desktopUa == enabled) return
        desktopUa = enabled
        applyUserAgent()
        webView.reload()
    }

    fun setBlockPopups(enabled: Boolean) {
        if (blockPopups == enabled) return
        blockPopups = enabled
        webView.settings.javaScriptCanOpenWindowsAutomatically = !enabled
    }

    fun destroy() {
        if (fullscreenContainer != null) {
            removeFullscreenInternal()
        }
        (webView.parent as? ViewGroup)?.removeView(webView)
        webView.removeJavascriptInterface("JqcBridge")
        webView.stopLoading()
        webView.loadUrl("about:blank")
        webView.clearHistory()
        webView.removeAllViews()
        webView.destroy()
    }

    private fun injectSniffer() {
        webView.evaluateJavascript(snifferScript, null)
    }

    private fun mobileUserAgent(): String =
        WebSettings.getDefaultUserAgent(activity).replace("; wv", "")

    private fun applyUserAgent() {
        webView.settings.userAgentString = if (desktopUa) DESKTOP_UA else mobileUserAgent()
    }

    private fun enterFullscreen(view: View, callback: WebChromeClient.CustomViewCallback) {
        if (fullscreenView != null) {
            callback.onCustomViewHidden()
            return
        }
        fullscreenView = view
        fullscreenCallback = callback

        val container = FrameLayout(activity).apply {
            setBackgroundColor(Color.BLACK)
            addView(view, FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT))
        }
        fullscreenContainer = container
        (activity.window.decorView as ViewGroup).addView(
            container,
            FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT)
        )

        previousOrientation = activity.requestedOrientation
        activity.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE
        activity.window.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
        WindowCompat.getInsetsController(activity.window, activity.window.decorView).apply {
            hide(WindowInsetsCompat.Type.systemBars())
            systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        }
    }

    fun isFullscreen(): Boolean = fullscreenContainer != null

    fun exitFullscreen() {
        if (fullscreenContainer == null) return
        val callback = fullscreenCallback
        removeFullscreenInternal()
        // 通知 WebView 页面退出全屏状态；若 WebView 同步回调 onHideCustomView，
        // 此时 fullscreenContainer 已为 null，会直接返回，不会重复移除。
        callback?.onCustomViewHidden()
    }

    private fun removeFullscreenInternal() {
        fullscreenContainer?.let {
            (activity.window.decorView as ViewGroup).removeView(it)
        }
        fullscreenContainer = null
        fullscreenView = null
        fullscreenCallback = null
        activity.requestedOrientation = previousOrientation
        activity.window.clearFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
        WindowCompat.getInsetsController(activity.window, activity.window.decorView)
            .show(WindowInsetsCompat.Type.systemBars())
    }

    private inner class Bridge {
        @JavascriptInterface
        fun reportMedia(url: String, hint: String) {
            if (url.length > 4096) return
            if (MediaUrlClassifier.detectType(url) == null) return
            val list = collector.offer(url, SniffSource.JS)
            activity.runOnUiThread { listener.onCandidates(list) }
        }
    }

    companion object {
        private const val DESKTOP_UA =
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36"
    }
}
```

```kotlin
package com.jqc.webplayer.player

import androidx.media3.common.PlaybackException

enum class PlayerErrorKind { HTTP, NETWORK, DECODING, UNKNOWN }

object PlayerErrorText {
    fun kindOf(errorCode: Int): PlayerErrorKind = when (errorCode) {
        PlaybackException.ERROR_CODE_IO_BAD_HTTP_STATUS,
        PlaybackException.ERROR_CODE_IO_FILE_NOT_FOUND,
        PlaybackException.ERROR_CODE_IO_INVALID_HTTP_CONTENT_TYPE,
        PlaybackException.ERROR_CODE_IO_NO_PERMISSION,
        -> PlayerErrorKind.HTTP

        PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_FAILED,
        PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_TIMEOUT,
        PlaybackException.ERROR_CODE_TIMEOUT,
        -> PlayerErrorKind.NETWORK

        PlaybackException.ERROR_CODE_DECODING_FAILED,
        PlaybackException.ERROR_CODE_DECODER_INIT_FAILED,
        PlaybackException.ERROR_CODE_DECODING_FORMAT_UNSUPPORTED,
        -> PlayerErrorKind.DECODING

        else -> PlayerErrorKind.UNKNOWN
    }

    fun ofKind(kind: PlayerErrorKind): String = when (kind) {
        PlayerErrorKind.HTTP -> "视频地址已失效或被拒绝（HTTP 错误）"
        PlayerErrorKind.NETWORK -> "网络连接失败"
        PlayerErrorKind.DECODING -> "设备不支持该视频编码"
        PlayerErrorKind.UNKNOWN -> "播放失败，请尝试用网页播放"
    }
}
```

```kotlin
package com.jqc.webplayer.player

import androidx.media3.common.C
import androidx.media3.common.Player
import androidx.media3.common.TrackGroup
import androidx.media3.common.TrackSelectionOverride

data class QualityOption(
    val label: String,
    val height: Int?,
    val group: TrackGroup?,
    val trackIndex: Int,
)

object TrackQualityController {
    fun qualities(player: Player): List<QualityOption> {
        val result = mutableListOf<QualityOption>()
        val tracks = player.currentTracks
        for (group in tracks.groups) {
            if (group.type != C.TRACK_TYPE_VIDEO) continue
            for (index in 0 until group.length) {
                val format = group.getTrackFormat(index)
                val height = format.height.takeIf { it > 0 }
                if (height != null) {
                    result += QualityOption(
                        label = "${height}P",
                        height = height,
                        group = group.mediaTrackGroup,
                        trackIndex = index,
                    )
                }
            }
        }
        return result.distinctBy { it.label }.sortedByDescending { it.height ?: 0 }
    }

    fun select(player: Player, option: QualityOption) {
        val group = option.group ?: return
        player.trackSelectionParameters = player.trackSelectionParameters.buildUpon()
            .setOverrideForType(TrackSelectionOverride(group, option.trackIndex))
            .build()
    }

    fun clearOverride(player: Player) {
        player.trackSelectionParameters = player.trackSelectionParameters.buildUpon()
            .clearOverridesOfType(C.TRACK_TYPE_VIDEO)
            .build()
    }
}
```

```kotlin
            android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE

        isInPictureInPictureMode: Boolean,
        newConfig: Configuration,

    }

    private fun hideSystemBars() {

            systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        }
    }

    override fun onStop() {
        saveProgress()
        super.onStop()
    }

    override fun onDestroy() {
        saveProgress()
        player.release()
        super.onDestroy()
    }

        url = intent.getStringExtra(PlayerContract.EXTRA_URL).orEmpty(),
        type = runCatching {

        }.getOrDefault(StreamType.HLS),
        pageUrl = intent.getStringExtra(PlayerContract.EXTRA_PAGE_URL).orEmpty(),
        title = intent.getStringExtra(PlayerContract.EXTRA_TITLE).orEmpty(),
        headers = PlayerContract.headersFrom(intent),
    )
```

```kotlin
class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            WebPlayerTheme {
                val container = (application as WebPlayerApp).container
                val history by container.historyRepository.observe()
                    .collectAsStateWithLifecycle(emptyList())
                val favorites by container.favoritesRepository.observe()
                    .collectAsStateWithLifecycle(emptyList())
                val settings by container.settingsRepository.flow
                    .collectAsStateWithLifecycle(AppSettings())
                val clipboardUrl = rememberClipboardUrl()
                val scope = rememberCoroutineScope()
                val hlsProbe = remember { HlsProbe() }
                var browserUrl by rememberSaveable { mutableStateOf<String?>(null) }
                var browserReloadToken by rememberSaveable { mutableIntStateOf(0) }

                val playerLauncher = rememberLauncherForActivityResult(
                    ActivityResultContracts.StartActivityForResult()
                ) { result ->
                    if (result.resultCode == PlayerContract.RESULT_OPEN_WEB) {
                        result.data
                            ?.getStringExtra(PlayerContract.EXTRA_PAGE_URL)
                            ?.takeIf { it.isNotBlank() }
                            ?.let { pageUrl ->
                                browserUrl = pageUrl
                                browserReloadToken += 1
                            }
                    }

                val currentUrl = browserUrl
                if (currentUrl == null) {
                    HomeScreen(
                        history = history,
                        favorites = favorites,
                        clipboardUrl = clipboardUrl,

                            browserUrl = url
                            browserReloadToken = 0
                        },

                            val videoUrl = entity.videoUrl

                            } else {

                                    StreamType.valueOf(entity.streamType ?: StreamType.HLS.name)
                                }.getOrDefault(StreamType.HLS)

                                        url = videoUrl,

                                browserUrl = entity.pageUrl
                            } else {
                                playerLauncher.launch(
                                    PlaybackRequest(
                                        url = videoUrl,
                                        type = StreamType.HLS,
                                        pageUrl = entity.pageUrl,
                                        title = entity.title,
                                        headers = emptyMap(),
                                    ).toIntent(this@MainActivity)

                        },
                        onDeleteFavorite = { id ->
```
